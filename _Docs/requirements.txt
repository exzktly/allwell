# requirements.txt
# -----------------
# Pip-only dependencies for AllWell — Apple Silicon macOS.
# Python 3.10.14, tensorflow-macos 2.13.0, tensorflow-metal 1.1.0
#
# ── ENVIRONMENT SETUP ────────────────────────────────────────────────────────
#
# Step 1 — create environment (setuptools pinned to <70 via mamba):
#
#   mamba create -n allwell python=3.10.14 setuptools=69.5.1 \
#       "numpy>=1.24,<1.25" scipy scikit-image matplotlib pillow imageio h5py \
#       -c conda-forge -y
#
# Step 2 — verify pkg_resources works:
#
#   mamba activate allwell
#   python -c "import pkg_resources; print('OK')"
#
# Step 3 — install pip packages:
#
#   pip install --no-build-isolation -r requirements.txt
#
# Step 4 — build the macOS app bundle:
#
#   scripts/build_executable.sh --clean
#
#   Always pass --clean. PyInstaller caches build/ and dist/ between runs
#   and will silently reuse stale TOCs that miss spec changes; --clean
#   wipes the cache and forces a fresh Analysis. The output lands at
#   ``dist/AllWell-<version>.app``, where <version> is the string in
#   version.py (a feature-subset build adds the profile name too).
#
# IMPORTANT:
#   - setuptools must stay at <70. Modern setuptools (v70+) removed
#     pkg_resources which is required by stardist and csbdeep.
#   - Always use --no-build-isolation with pip to prevent it from
#     escaping to the system Python for wheel builds.
#   - Never run: pip install setuptools  (this will upgrade and break things)
#
# ── VERIFY GPU AFTER INSTALL ─────────────────────────────────────────────────
#
#   python -c "import tensorflow as tf; print(tf.config.list_physical_devices('GPU'))"
#   # Expected: [PhysicalDevice(name='/physical_device:GPU:0', device_type='GPU')]
#
# ── ENVIRONMENT SETUP (Windows, native CPU) ──────────────────────────────────
#
# Native Windows TF has been CPU-only since 2.11 (TF 2.10 was the last release
# with native Windows GPU). We pin tensorflow==2.13.0 to keep the
# keras/protobuf/StarDist stack identical across platforms. For GPU on
# Windows, use WSL2 (out of scope here).
#
# Use Miniforge so `mamba` is available on Windows. From the "Miniforge
# Prompt" (or PowerShell with `mamba init powershell` run once), in the
# repo root.
#
# IMPORTANT: If you previously created the `allwell` env using older
# instructions that listed `pyside6` in the mamba step, remove it first.
# Conda-installed shiboken6/pyside6 cannot be uninstalled by pip when
# the `PySide6==6.7.3` pin below tries to downgrade them:
#
#   mamba env remove -n allwell -y
#
# Then create the env (one line — bash `\` does not work in PowerShell
# or in the Miniforge Prompt):
#
#   mamba create -n allwell python=3.10.14 setuptools=69.5.1 "numpy>=1.24,<1.25" scipy scikit-image matplotlib-base pillow imageio h5py vs2015_runtime -c conda-forge -y
#   mamba activate allwell
#   pip install --no-build-isolation -r _Docs\requirements.txt
#   python allwell.py
#
# To produce a frozen Windows bundle (optional; only for distribution):
#
#   powershell -ExecutionPolicy Bypass -File scripts\build_executable.ps1 -Clean
#   # produces dist\AllWell-<version>\AllWell.exe, where <version> is the
#   # string in version.py. -ExecutionPolicy Bypass is scoped to that one
#   # process; a stock Windows install refuses to run an unsigned .ps1
#   # otherwise, and the Miniforge Prompt is cmd.exe, which cannot run one
#   # at all.
#
# Notes:
#   - The mamba step pulls in `vs2015_runtime` from conda-forge so the
#     MSVC redistributable DLLs PySide6 needs are available without
#     admin rights. Do NOT add `pyside6` to the mamba step — PySide6
#     and shiboken6 are installed by pip from the pinned versions
#     below. Conda-installed Qt packages cannot be uninstalled by pip
#     (no RECORD file), so a mixed install will fail with
#     `Cannot uninstall Shiboken6 6.x.y` on the next pip run.
#   - The mamba step uses `matplotlib-base` (not `matplotlib`) on
#     purpose. The conda-forge `matplotlib` meta-package pulls in
#     `pyside6` (its preferred Qt binding) as a transitive dep, which
#     drags in the latest `shiboken6` — same uninstall failure as
#     above. `matplotlib-base` is all of matplotlib without the Qt
#     binding; PySide6 from pip supplies the Qt backend at runtime.
#   - The command is a single line on purpose. Bash-style `\`
#     continuations do NOT work in PowerShell (use `` ` ``) or in
#     the Miniforge Prompt / cmd.exe (use `^`). Keeping it on one
#     line works in all three.
#   - numba/llvmlite Windows wheels bundle the LLVM runtime; no separate
#     MSVC install needed for them.
#   - tensorflow==2.13.0 has Windows wheels for Python 3.10 and 3.11
#     only. Python 3.12+ will fail at the pip step.
#   - The same setuptools<70 / pkg_resources rationale documented above
#     applies on Windows.
#
# ── VERIFY CPU AFTER INSTALL (Windows) ───────────────────────────────────────
#
#   python -c "import tensorflow as tf; print(tf.config.list_physical_devices('GPU'))"
#   # Expected: []
#   python -c "import tensorflow as tf; print(tf.config.list_physical_devices('CPU'))"
#   # Expected: [PhysicalDevice(name='/physical_device:CPU:0', device_type='CPU')]

# ── Pipeline + scientific stack (shared with the headless remote) ────────────
# v5.2.1: split off so the bootstrap can install just the pipeline deps
# on the remote without dragging PySide6/pyobjc/pyinstaller through
# (some of which lack wheels on Linux aarch64 / newer Pythons). See
# _Docs/requirements-remote.txt for the full list.
#
# This install targets StarDist + Cellpose *inference* only.
-r requirements-remote.txt

# ── GUI toolkit (Qt for Python) ──────────────────────────────────────────────
# Replaces tkinter. Pulls in shiboken6 transitively at the matching version.
# On Windows, prefer `mamba install -c conda-forge pyside6` so the MSVC
# runtime DLLs are provided by the `vc`/`vs2015_runtime` conda packages
# (avoids needing admin to install the system vcredist).
PySide6==6.7.3

# ── Multivariate (higher-dimensional analysis) — Review-side only ────────────
# The Multivariate tab's UMAP / t-SNE projections. PCA / correlation /
# k-means / hierarchical need nothing beyond numpy+scipy (already pinned); these
# two add the non-linear methods. Pinned to versions that COEXIST with the
# binding caps the segmentation stack imposes — verified by a full resolver
# dry-run: scikit-learn 1.3.2 + umap-learn 0.5.5 keep numpy at 1.24.3 (<1.25 for
# TF 2.13 / numba 0.57) and typing-extensions at 4.5.0 (<4.6 for TF 2.13); their
# only extra transitives (pynndescent, joblib, threadpoolctl, tqdm) are already
# numba/cellpose-compatible. NOT in requirements-remote.txt — the headless
# pipeline never runs Review analysis. Before bumping either pin, re-run
# `pip install --dry-run -r requirements.txt` and confirm numpy /
# typing-extensions / numba are not perturbed.
scikit-learn==1.3.2
umap-learn==0.5.5

# ── Per-well frame cache (Review-side only) ──────────────────────────────────
# The Parquet cache beside each measurements CSV
# (well_viewer/data/parquet_cache.py). Measured through the app's own read path
# on a 120,000-row x 37-column well: 3,068 ms to parse the CSV against 51 ms for
# the cached frame, and 7 ms for the six columns a plot needs.
#
# It is OPTIONAL at runtime — every entry point in that module answers "no" when
# pyarrow is absent and the CSV is parsed exactly as it always was — so a
# partial environment loses speed and nothing else. NOT in
# requirements-remote.txt: the cache is a Review-side read, the pipeline writes
# CSVs and never reads one back.
#
# It is the one dependency here with a real cost, and the number is measured
# rather than recalled: the cp311 manylinux wheel is 48 MB compressed and
# 155 MB unpacked (159 MB as installed on this machine). PyInstaller strips and
# recompresses, so the .app grows by less than that — by how much has NOT been
# measured, and the unpacked size is the honest upper bound to quote. The trade
# is stated rather than assumed: five minutes of parsing per plate open, every
# open, against that.
#
# Pinned like every other line in this file rather than left as a range. It was
# `>=14.0`, which resolves to whatever is newest on the day and makes the
# "verified not to perturb the caps" note below true only of the version that
# happened to be installed when it was written. pyarrow 22.0.0's wheel declares
# no `Requires-Dist` at all, so it cannot move numpy or pandas; verified
# installed alongside numpy 1.24.4 and pandas 2.2.3 with the fast lane green.
pyarrow==22.0.0

# ── macOS native clipboard (Copy SVG) ────────────────────────────────────────
# Qt's QMimeData cannot register a payload under a raw macOS UTI, so Copy SVG
# would only ever paste raster into Illustrator. pyobjc lets us write
# NSPasteboard directly under com.adobe.pdf / public.svg-image. macOS-only;
# off macOS Copy SVG falls back to the Qt clipboard.
pyobjc-framework-Cocoa==10.3.1 ; sys_platform == "darwin"

# ── Build tooling (build only) ───────────────────────────────────────────────
pyinstaller==6.1.0
