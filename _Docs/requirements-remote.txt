# requirements-remote.txt
# ------------------------
# Pipeline-only Python dependencies for the AllWell *remote* execution
# host (a Linux machine the user SSHes into). The bootstrap script
# (scripts/_bootstrap_lib.py) pip-installs this file inside the
# ``allwell`` conda env on the remote.
#
# Why a separate file from requirements.txt? The GUI side ships PySide6,
# pyobjc, and pyinstaller — none of which the headless pipeline needs.
# Trying to pip-install PySide6 on certain Pythons / architectures
# (notably Linux aarch64) fails outright because PySide6 doesn't ship
# wheels for those platforms. v5.2.1 splits the GUI superset off so the
# remote install never touches PySide6.
#
# All version pins below are identical to the GUI requirements file so
# the pipeline's scientific output is bit-identical across the local and
# remote hosts (StarDist segmentation determinism depends on it).

# ── TensorFlow (Apple Silicon, GPU via Metal) ────────────────────────────────
tensorflow-macos==2.13.0 ; sys_platform == "darwin" and platform_machine == "arm64"
tensorflow-metal==1.1.0  ; sys_platform == "darwin" and platform_machine == "arm64"

# ── TensorFlow (Linux x86_64 + Windows, CPU-only) ────────────────────────────
# Pin 2.13.0 to keep keras / protobuf / StarDist identical across platforms.
tensorflow==2.13.0 ; sys_platform != "darwin"

# Shared by both TF lines (cross-platform pins)
keras==2.13.1
protobuf==3.20.3

# ── StarDist segmentation pipeline ───────────────────────────────────────────
csbdeep==0.8.0
stardist==0.9.1
numba==0.57.1
llvmlite==0.40.1

# ── Image I/O ────────────────────────────────────────────────────────────────
tifffile==2023.9.26
imagecodecs==2023.9.18

# ── Data analysis ────────────────────────────────────────────────────────────
# Used by post-pipeline CSV assembly; minimal footprint compared to the
# scientific stack.
pandas==2.2.3

# ── TOML config (used by services/remote_config.py on Python 3.10) ───────────
tomli ; python_version < "3.11"

# ── Test environment guard (same as the GUI requirements) ────────────────────
# typing-extensions pin protection — see _Docs/requirements.txt header
# comment for the gory details.
exceptiongroup<1.3
