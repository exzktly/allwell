# requirements.txt
# -----------------
# Pip-only dependencies for AllWell — Apple Silicon macOS.
# Python 3.10.14, tensorflow-macos 2.13.0, tensorflow-metal 1.1.0
#
# ── ENVIRONMENT SETUP ────────────────────────────────────────────────────────
#
# Step 1 — create environment (setuptools pinned to <70 via mamba):
#
#   mamba create -n allwell python=3.10.14 setuptools=69.5.1 \
#       "numpy>=1.24,<1.25" scipy scikit-image matplotlib pillow imageio h5py \
#       -c conda-forge -y
#
# Step 2 — verify pkg_resources works:
#
#   mamba activate allwell
#   python -c "import pkg_resources; print('OK')"
#
# Step 3 — install pip packages:
#
#   pip install --no-build-isolation -r requirements.txt
#
# Step 4 — build the macOS app bundle:
#
#   scripts/build_executable.sh --clean
#
#   Always pass --clean. PyInstaller caches build/ and dist/ between runs
#   and will silently reuse stale TOCs that miss spec changes; --clean
#   wipes the cache and forces a fresh Analysis. The output lands at
#   ``dist/AllWell.app``.
#
# IMPORTANT:
#   - setuptools must stay at <70. Modern setuptools (v70+) removed
#     pkg_resources which is required by stardist and csbdeep.
#   - Always use --no-build-isolation with pip to prevent it from
#     escaping to the system Python for wheel builds.
#   - Never run: pip install setuptools  (this will upgrade and break things)
#
# ── VERIFY GPU AFTER INSTALL ─────────────────────────────────────────────────
#
#   python -c "import tensorflow as tf; print(tf.config.list_physical_devices('GPU'))"
#   # Expected: [PhysicalDevice(name='/physical_device:GPU:0', device_type='GPU')]
#
# ── ENVIRONMENT SETUP (Windows, native CPU) ──────────────────────────────────
#
# Native Windows TF has been CPU-only since 2.11 (TF 2.10 was the last release
# with native Windows GPU). We pin tensorflow==2.13.0 to keep the
# keras/protobuf/StarDist stack identical across platforms. For GPU on
# Windows, use WSL2 (out of scope here).
#
# Use Miniforge so `mamba` is available on Windows. From the "Miniforge
# Prompt" (or PowerShell with `mamba init powershell` run once), in the
# repo root:
#
#   mamba create -n allwell python=3.10.14 setuptools=69.5.1 "numpy>=1.24,<1.25" scipy scikit-image matplotlib pillow imageio h5py vs2015_runtime -c conda-forge -y
#   mamba activate allwell
#   pip install --no-build-isolation -r _Docs\requirements.txt
#   python all_well.py
#
# To produce a frozen Windows bundle (optional; only for distribution):
#
#   powershell scripts\build_executable.ps1 -Clean   # produces dist\AllWell\
#
# Notes:
#   - The mamba step pulls in `vs2015_runtime` from conda-forge so the
#     MSVC redistributable DLLs PySide6 needs are available without
#     admin rights. PySide6 itself is installed by pip from the
#     `PySide6==6.7.3` pin below — installing it via conda first
#     would be redundant (pip would reinstall it from PyPI and clobber
#     the conda build).
#   - The command is a single line on purpose. Bash-style `\`
#     continuations do NOT work in PowerShell (use `` ` ``) or in
#     the Miniforge Prompt / cmd.exe (use `^`). Keeping it on one
#     line works in all three.
#   - numba/llvmlite Windows wheels bundle the LLVM runtime; no separate
#     MSVC install needed for them.
#   - tensorflow==2.13.0 has Windows wheels for Python 3.10 and 3.11
#     only. Python 3.12+ will fail at the pip step.
#   - The same setuptools<70 / pkg_resources rationale documented above
#     applies on Windows.
#
# ── VERIFY CPU AFTER INSTALL (Windows) ───────────────────────────────────────
#
#   python -c "import tensorflow as tf; print(tf.config.list_physical_devices('GPU'))"
#   # Expected: []
#   python -c "import tensorflow as tf; print(tf.config.list_physical_devices('CPU'))"
#   # Expected: [PhysicalDevice(name='/physical_device:CPU:0', device_type='CPU')]

# ── Pipeline + scientific stack (shared with the headless remote) ────────────
# v5.2.1: split off so the bootstrap can install just the pipeline deps
# on the remote without dragging PySide6/pyobjc/pyinstaller through
# (some of which lack wheels on Linux aarch64 / newer Pythons). See
# _Docs/requirements-remote.txt for the full list.
-r requirements-remote.txt

# ── GUI toolkit (Qt for Python) ──────────────────────────────────────────────
# Replaces tkinter. Pulls in shiboken6 transitively at the matching version.
# On Windows, prefer `mamba install -c conda-forge pyside6` so the MSVC
# runtime DLLs are provided by the `vc`/`vs2015_runtime` conda packages
# (avoids needing admin to install the system vcredist).
PySide6==6.7.3

# ── macOS native clipboard (Copy SVG) ────────────────────────────────────────
# Qt's QMimeData cannot register a payload under a raw macOS UTI, so Copy SVG
# would only ever paste raster into Illustrator. pyobjc lets us write
# NSPasteboard directly under com.adobe.pdf / public.svg-image. macOS-only;
# off macOS Copy SVG falls back to the Qt clipboard.
pyobjc-framework-Cocoa==10.3.1 ; sys_platform == "darwin"

# ── Build tooling (build only) ───────────────────────────────────────────────
pyinstaller==6.1.0
